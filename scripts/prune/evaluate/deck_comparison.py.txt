"""Render the report-10-19 checkpoint comparison without changing a frozen study.

Run in ltx from LTX-2. See doc/evaluate/deck_comparison.md. This separate
executor uses saved source/noise, actual gates and adapters, and replay controls.
The .txt suffix preserves the stopped studies' frozen Python inventories.
"""

import argparse
import copy
import hashlib
import json
from pathlib import Path

import torch

from scripts.onestep_avatar import media
from scripts.onestep_avatar.corpus.dataset import load_training_master
from scripts.prune.core import ltx_adapter, provenance, session
from scripts.prune.data import whole_clip
from scripts.prune.evaluate.mobilewan_media import source_pixels
from scripts.prune.learn import adapters, checkpoints
from scripts.prune.learn.gates import HeadGates
from scripts.prune.learn.heads import numerical_policy
from scripts.prune.learn.progressive import ProgressivePolicy


def read(path):
    return json.loads(Path(path).read_text())


def pin(path):
    return dict(path=str(Path(path).resolve()), sha256=provenance.file_sha256(path))


def sample(model, grid, video, c0, schedule):
    """Use the stock FP32 Euler step and clamp the clean first frame each time."""
    from dataclasses import replace
    from ltx_core.components.diffusion_steps import EulerDiffusionStep
    from scripts.onestep_avatar.model import common

    sigmas = torch.tensor(schedule, device=video.latent.device, dtype=torch.float32)
    value = video.latent
    stepper = EulerDiffusionStep()
    for i in range(len(schedule) - 1):
        level = sigmas[i:i + 1]
        times = torch.where(video.timesteps > 0, level, 0).to(video.timesteps.dtype)
        modality = replace(video, latent=value, sigma=level, timesteps=times)
        if i == 0 and not torch.equal(modality.timesteps, video.timesteps):
            raise ValueError("initial Euler timesteps differ from saved input")
        _, clean = ltx_adapter.velocity_forward(model, modality)
        clean = common.with_clean_prefix(clean, c0)
        value = common.with_clean_prefix(stepper.step(value, clean, sigmas, i), c0)
    return grid.unpatchify_block(value, grid.latent_frames).cpu()


def inputs(study, source, current, start):
    key = hashlib.sha256(source['relative_dir'].encode()).hexdigest()
    folder = Path(study['root']) / f'inputs/{key}/seed_2042'
    manifest = whole_clip.load_manifest(folder)
    # Only the schedule is extended. All source and epsilon bytes stay pinned.
    local = copy.deepcopy(manifest)
    if start not in local['sigmas']:
        local['sigmas'].append(start)
        row = copy.deepcopy(local['videos'][0])
        row.update(sigma=start, schedule=[start, 0.0])
        local['videos'].append(row)
    grid, video, c0, row = whole_clip.build_input(
        folder, local, view=source['resolved_dir'], sigma=start, current=current)
    evidence = dict(manifest=pin(folder / 'manifest.json'), capture=pin(source['capture']),
                    epsilon_file_sha256=row['artifacts']['epsilon_sha256'],
                    epsilon_tensor_sha256=checkpoints.tensor_hash(whole_clip.load_epsilon(folder, row)),
                    context_tensor_sha256=checkpoints.tensor_hash(current.context),
                    noisy_tokens_sha256=checkpoints.tensor_hash(video.latent),
                    clean_first_frame_sha256=checkpoints.tensor_hash(c0),
                    source=source, geometry=manifest['geometry'], fps=source['fps'])
    return grid, video, c0, evidence


def configure(model, attached, state, name, device):
    if name == 'native':
        for module in attached.values():
            module.enabled = False
        return None
    adapters.load_state(attached, state['adapters'])
    for module in attached.values():
        module.enabled = True
    clipped = name == 'new'
    gates = HeadGates(48, 32, activation='clip01_v1' if clipped else 'sigmoid_v1').to(device)
    gates.alpha.data.copy_(state['alpha'].to(device))
    gates.progressive = ProgressivePolicy.from_state(state['progressive'])
    gates.phase = 'recovery' if clipped and gates.progressive.recovering else 'soft'
    gates.microbatch()
    if int((gates.current == 0).sum()) != (292 if clipped else 867):
        raise ValueError('loaded checkpoint has the wrong removal count')
    return gates


@torch.no_grad()
def run(args):
    numerical_policy()
    torch.set_num_threads(8)
    root = args.output.resolve()
    old = args.old.resolve()
    new = args.new.resolve()
    study = read(new / 'runs/02_continuation_u500-847/configs/frozen_study.json')
    source = next(s for s in study['splits']['validation'] if s['actor'] == args.actor)
    grid_values = study['schedule']['original_grid']
    tail = grid_values[grid_values.index(args.start):]
    schedule = ([tail[round(j * (len(tail)-1) / 4)] for j in range(5)]
                if args.schedule == 'four' else tail)
    case = root / f'actor_{args.actor}/{args.schedule}'
    case.mkdir(parents=True, exist_ok=True)
    old_stage = next((old / 'stages').glob('04_*'))
    checkpoint_paths = dict(old=old_stage / 'training/learned_progressive/replicate_42/checkpoints/update_3300.pt',
                            new=new / 'runs/02_continuation_u500-847/training/learned_progressive/replicate_42/checkpoints/update_0800.pt')
    current = session.open_session(argparse.Namespace(model='2.5', gpu_id=0, seed=42),
                                   script='prune.evaluate.deck_comparison', prompt='',
                                   transformer_path=Path(study['components']['transformer']['path']))
    if checkpoints.tensor_hash(current.context) != study['text_context']['tensor_sha256']:
        raise ValueError('context differs from frozen study')
    grid, video, c0, evidence = inputs(study, source, current, args.start)
    receipts, controls = {}, []
    old_videos = read(old / 'final_native/media.json')['videos']
    reuse = next((r for r in old_videos if r['kind'] == 'steps' and r['actor'] == args.actor
                  and r['schedule'] == schedule), None)
    if reuse:
        matches = read(old / 'final_native/matched_inputs.json')['cases']
        paired = next(r for r in matches if r['actor'] == args.actor)
        if (paired['pins']['capture'] != evidence['capture']['sha256']
                or paired['pins']['epsilon'] != evidence['epsilon_file_sha256']):
            raise ValueError('archived source or noise differs from the new comparison')
        for name, pos in [('native', 1), ('old', 2)]:
            p = old / reuse['sources'][pos]['path']
            if provenance.file_sha256(p) != reuse['sources'][pos]['sha256']:
                raise ValueError('archived video changed')
            receipts[name] = dict(video=pin(p), reused=True)
    states = {}
    needed = [name for name in ('native', 'old', 'new') if name not in receipts]
    for name in ('old', 'new'):
        if name in needed:
            state = checkpoints.load(checkpoint_paths[name])
            states[name] = {k: state[k] for k in ('adapters', 'alpha', 'progressive')}
            del state
    with current.transformer(Path(study['components']['transformer']['path'])) as model:
        model.eval().requires_grad_(False)
        attached = adapters.attach(model, 42, 128)
        for name in needed:
            gates = configure(model, attached, states.get(name), name, current.device)
            handles = None if gates is None else __import__(
                'scripts.prune.score.hooks', fromlist=['attach_head_provider']).attach_head_provider(model, gates.provider)
            try:
                # Replay each trained policy at its saved monitor noise level.
                if name in ('old', 'new'):
                    control_sigma = 0.9489720463752747
                    cg, cv, cc, _ = inputs(study, source, current, control_sigma)
                    _, clean = ltx_adapter.velocity_forward(model, cv)
                    from scripts.onestep_avatar.model import common
                    actual = cg.unpatchify_block(common.with_clean_prefix(clean, cc), cg.latent_frames).cpu()
                    euler = sample(model, cg, cv, cc, [control_sigma, 0.0])
                    reduction = float((euler.float() - actual.float()).abs().max())
                    controls.append(dict(method=name, control='Euler single-step versus direct',
                                         max_abs=reduction, tolerance=0.02, passed=reduction <= 0.02))
                    if reduction > 0.02:
                        raise ValueError('Euler sampler single-step reduction failed')
                    del euler
                    if name == 'new':
                        owner = new / 'runs/02_continuation_u500-847'
                        monitor = read(owner / 'checks/learned_progressive/replicate_42/monitor_0800.json')
                        ref = next(r for r in monitor['rows'] if r['actor'] == args.actor and r['sigma'] == control_sigma)
                        expected = Path(ref['states']['trained']['path'])
                    else:
                        owner = next((old / 'stages').glob('04_*'))
                        monitor = read(owner / 'checks/learned_progressive/replicate_42/monitor_3300.json')
                        ref = next(r for r in monitor['rows'] if r['actor'] == args.actor and r['sigma'] == control_sigma)
                        original = ref['states']['trained']['path']
                        archive = read(old / 'provenance/archive_manifest.json')
                        expected = old / archive['path_map'][original]
                    if provenance.file_sha256(expected) != ref['states']['trained']['sha256']:
                        raise ValueError('saved monitor bytes changed')
                    target = torch.load(expected, map_location='cpu', weights_only=True)
                    delta = float((actual.float() - target.float()).abs().max())
                    control = dict(method=name, max_abs=delta, tolerance=0.02,
                                   bit_identical=torch.equal(actual, target), passed=delta <= 0.02)
                    controls.append(control)
                    print('Replay', args.actor, control, flush=True)
                    if not control['passed']:
                        raise ValueError('checkpoint replay failed')
                    del cg, cv, cc, clean, actual, target
                dest = case / f'{name}.pt'
                meta = dest.with_suffix('.json')
                identity = dict(method=name, schedule=schedule, inputs=evidence,
                                checkpoint=None if name == 'native' else pin(checkpoint_paths[name]),
                                executor=pin(__file__))
                if meta.exists():
                    previous = read(meta)
                    if previous['identity'] != identity or pin(dest) != previous['latent']:
                        raise ValueError('existing prediction identity changed')
                else:
                    print('Sample', args.actor, name, args.schedule, len(schedule)-1, flush=True)
                    latent = sample(model, grid, video, c0, schedule)
                    checkpoints.publish_tensor(dest, latent)
                    checkpoints.publish_json(meta, dict(identity=identity, latent=pin(dest)))
                    del latent
                receipts[name] = dict(latent=pin(dest), reused=False)
            finally:
                if handles is not None:
                    handles.detach_all()
        del model, attached, gates, states, video, c0
    torch.cuda.empty_cache()
    from ltx_trainer.video_utils import save_video
    with current.decoder() as decoder:
        for name, item in receipts.items():
            if 'video' in item:
                continue
            latent = torch.load(item['latent']['path'], map_location='cpu', weights_only=True)
            pixels = media.decode(current, latent, decoder, 42).mul(255).round().to(torch.uint8)
            dest = case / f'{name}.mp4'
            save_video(pixels, dest, fps=source['fps'], video_format='FCHW')
            item['video'] = pin(dest)
            del latent, pixels
            print('Decoded', args.actor, name, args.schedule, flush=True)
        gt_path = root / f'actor_{args.actor}/capture_vae.mp4'
        if not gt_path.exists():
            capture, _ = load_training_master(Path(source['capture']))
            pixels = media.decode(current, capture.unsqueeze(0), decoder, 42).mul(255).round().to(torch.uint8)
            save_video(pixels, gt_path, fps=source['fps'], video_format='FCHW')
            del capture, pixels
    rgb_path = root / f'actor_{args.actor}/capture_rgb.mp4'
    if not rgb_path.exists():
        save_video(source_pixels(source), rgb_path, fps=source['fps'], video_format='FCHW')
    checkpoints.publish_json(case / 'receipt.json', dict(
        actor=args.actor, schedule=schedule, steps=len(schedule)-1, inputs=evidence,
        outputs=receipts, controls=controls, capture_vae=pin(gt_path), capture_rgb=pin(rgb_path),
        decoder=media.native_decoder_settings(), decoder_seed=42, physical_gpu=args.gpu,
        executor=pin(__file__), passed=True))
    print('COMPLETE', case, flush=True)


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--old', type=Path, required=True)
    parser.add_argument('--new', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--actor', required=True, choices=['35', '25', '84'])
    parser.add_argument('--schedule', required=True, choices=['four', 'tail'])
    parser.add_argument('--start', type=float, required=True)
    parser.add_argument('--gpu', type=int, required=True, choices=range(8))
    arguments = parser.parse_args()
    import os
    if os.environ.get('CUDA_VISIBLE_DEVICES') != str(arguments.gpu):
        raise ValueError('set CUDA_VISIBLE_DEVICES to the one authorized physical GPU')
    run(arguments)
